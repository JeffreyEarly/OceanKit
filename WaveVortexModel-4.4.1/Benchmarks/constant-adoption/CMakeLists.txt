cmake_minimum_required(VERSION 3.20)
project(WVConstantAdoption LANGUAGES CXX)
set(WVM_SOURCE "" CACHE PATH "Frozen control or candidate WVM checkout")
if(NOT EXISTS "${WVM_SOURCE}/PortableRuntime/CMakeLists.txt")
    message(FATAL_ERROR "WVM_SOURCE must identify an authoring checkout")
endif()
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(WV_RUNTIME_BUILD_BENCHMARKS OFF CACHE BOOL "" FORCE)
set(WV_RUNTIME_ENABLE_NATIVE_FFTW ON CACHE BOOL "" FORCE)
add_subdirectory("${WVM_SOURCE}/PortableRuntime" runtime)
add_executable(wv-constant-adoption WVConstantAdoptionWorker.cpp)
target_compile_features(wv-constant-adoption PRIVATE cxx_std_17)
target_include_directories(wv-constant-adoption PRIVATE "${WVM_SOURCE}/PortableRuntime/src" "${WVM_SOURCE}/CompiledKernel/adapters/native-fftw")
target_link_libraries(wv-constant-adoption PRIVATE WaveVortex::PortableRuntime WaveVortexNativeFFTW)
execute_process(COMMAND git -C "${WVM_SOURCE}" rev-parse HEAD OUTPUT_VARIABLE adoption_source OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
target_compile_definitions(wv-constant-adoption PRIVATE WV_RUNTIME_EXPECTED_FFTW_ROOT="${WV_RUNTIME_FFTW_ROOT}" WV_RUNTIME_SOURCE_COMMIT="${adoption_source}")
set_target_properties(wv-constant-adoption PROPERTIES CXX_EXTENSIONS OFF BUILD_RPATH "${WV_RUNTIME_FFTW_ROOT}/lib")
if(CMAKE_CXX_COMPILER_ID MATCHES "AppleClang|Clang|GNU")
    target_compile_options(wv-constant-adoption PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
