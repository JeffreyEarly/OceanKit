cmake_minimum_required(VERSION 3.20)
project(WVVariableAdoption LANGUAGES CXX)
set(WVM_SOURCE "" CACHE PATH "WVM authoring checkout")
if(NOT EXISTS "${WVM_SOURCE}/PortableRuntime/CMakeLists.txt")
    message(FATAL_ERROR "WVM_SOURCE must identify an authoring checkout")
endif()
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(WV_RUNTIME_BUILD_BENCHMARKS OFF CACHE BOOL "" FORCE)
set(WV_RUNTIME_ENABLE_NATIVE_FFTW ON CACHE BOOL "" FORCE)
add_subdirectory("${WVM_SOURCE}/PortableRuntime" runtime)
add_executable(wv-variable-adoption WVVariableAdoptionWorker.cpp)
add_executable(wv-variable-derivative WVVariableDerivativeWorker.cpp)
target_compile_features(wv-variable-adoption PRIVATE cxx_std_17)
target_compile_features(wv-variable-derivative PRIVATE cxx_std_17)
target_include_directories(wv-variable-adoption PRIVATE "${WVM_SOURCE}/PortableRuntime/third_party")
target_include_directories(wv-variable-derivative PRIVATE "${WVM_SOURCE}/PortableRuntime/third_party")
target_link_libraries(wv-variable-adoption PRIVATE WaveVortex::PortableRuntime WaveVortexNativeFFTW WaveVortex::AccelerateMatrix)
target_link_libraries(wv-variable-derivative PRIVATE WaveVortex::PortableRuntime WaveVortexNativeFFTW WaveVortex::AccelerateMatrix)
set_target_properties(wv-variable-adoption PROPERTIES CXX_EXTENSIONS OFF BUILD_RPATH "${WV_RUNTIME_FFTW_ROOT}/lib")
set_target_properties(wv-variable-derivative PROPERTIES CXX_EXTENSIONS OFF BUILD_RPATH "${WV_RUNTIME_FFTW_ROOT}/lib")
if(CMAKE_CXX_COMPILER_ID MATCHES "AppleClang|Clang|GNU")
    target_compile_options(wv-variable-adoption PRIVATE -Wall -Wextra -Wpedantic -Werror)
    target_compile_options(wv-variable-derivative PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
